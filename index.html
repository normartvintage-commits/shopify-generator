function sendToChat() { const title = buildTitle(); if (!title || title.indexOf('←') !== -1) { showToast('⚠️ ブランドまたはカテゴリを入力してください'); return; } const priceRaw = document.getElementById('price').value.replace(/,/g, ''); const price = priceRaw ? String(Math.round(parseFloat(priceRaw))) : '0'; const inventoryEl = document.getElementById('inventory'); const inventory = inventoryEl && inventoryEl.value.trim() ? inventoryEl.value.trim() : ''; const tags = buildTags(); const descHtml = buildHtml(); const category = document.getElementById('mf-category').value; const countryMF = document.getElementById('mf-country').value; const brand = document.getElementById('mf-brand').value; const topVal = document.getElementById('item-no-top').value.trim(); const itemNos = topVal ? topVal.split(',').map(s => s.trim()).filter(Boolean) : []; const statusEl = document.getElementById('claude-status'); statusEl.style.display = 'block'; statusEl.style.background = 'var(--bg)'; statusEl.style.color = 'var(--text-muted)'; statusEl.innerHTML = '⏳ Shopifyに登録中...'; const btn = document.getElementById('claude-register-btn'); if (btn) btn.disabled = true; const payload = { title: title, price: parseInt(price) || 0, inventory: inventory ? parseInt(inventory) : 0, body_html: descHtml, brand: brand, category: category, country: countryMF, tags: tags, itemNos: itemNos }; const callbackName = 'gasRegisterCb_' + Date.now(); const script = document.createElement('script'); script.src = GAS_URL + '?action=register&callback=' + callbackName + '&payload=' + encodeURIComponent(JSON.stringify(payload)); window[callbackName] = function(result) { delete window[callbackName]; try { document.body.removeChild(script); } catch (e) {} if (btn) btn.disabled = false; if (result && result.success) { statusEl.style.background = 'var(--success-bg)'; statusEl.style.color = 'var(--success)'; statusEl.innerHTML = '✅ ' + result.message + '
Shopify管理画面で開く'; showToast('✅ 登録完了！'); } else { statusEl.style.background = 'var(--warning-bg)'; statusEl.style.color = 'var(--warning)'; statusEl.innerHTML = '⚠️ エラー: ' + ((result && result.error) || '不明なエラー'); showToast('⚠️ 登録に失敗しました'); } }; script.onerror = function() { try { document.body.removeChild(script); } catch (e) {} delete window[callbackName]; if (btn) btn.disabled = false; statusEl.style.background = 'var(--warning-bg)'; statusEl.style.color = 'var(--warning)'; statusEl.innerHTML = '⚠️ 接続エラー'; }; setTimeout(function() { if (window[callbackName]) { delete window[callbackName]; try { document.body.removeChild(script); } catch (e) {} if (btn) btn.disabled = false; statusEl.style.background = 'var(--warning-bg)'; statusEl.style.color = 'var(--warning)'; statusEl.innerHTML = '⚠️ タイムアウト'; } }, 60000); document.body.appendChild(script); }
